# AI policy: search and answer engines may crawl and cite this site; training
# models on it is not allowed. AI search and user-triggered fetchers
# (OAI-SearchBot, ChatGPT-User, Claude-SearchBot, Claude-User, PerplexityBot,
# Perplexity-User, Googlebot, Bingbot) follow the * group below. Training-only
# crawlers get their own group further down.
User-agent: *
Content-Signal: search=yes, ai-input=yes, ai-train=no
# Internal plumbing and pages with no indexable content of their own.
Disallow: /api/
Disallow: /_a/
# /_proxy-api/ is NOT disallowed: entity tables load through it, and Google
# renders a page only with the fetches robots.txt lets it make. Blocking it
# left crawlers an empty table. Its responses carry X-Robots-Tag: noindex, and
# the proxy's route allowlist (not this file) is what limits it.
# /results is crawlable on purpose, so Google reads its noindex.
Disallow: /report/
# Chemical composition rows link to /food/<x>?highlight=<chemical id>: one URL
# per food-chemical pair, ~200k variants of pages the sitemap already lists
# bare. The canonical dedupes them only after they have been fetched.
# Both positions, since robots patterns are anchored at the path.
Disallow: /*?highlight=
Disallow: /*&highlight=
# Last, so parsers that take the first matching rule (not the longest) still
# apply the Disallows above.
Allow: /
